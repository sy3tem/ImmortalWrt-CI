# armsr/armv8 — UEFI（edk2 / AAVMF）虚拟机镜像
#
# 目标：ImmortalWrt 官方 armsr/armv8 (Generic EFI Boot)，产物是标准 GPT 磁盘：
#   p1 = FAT ESP（GRUB，BOOTAA64.EFI）+ p2 = ext4 rootfs
# 在 fnOS（飞牛）虚拟机上用 UEFI 固件 + virtio 磁盘/网卡直接启动，不需要 U-Boot。
#
# 网口数量在首次开机时自动识别（target-patch/armsr/files/etc/uci-defaults/92-nic-autoconfig）：
#   1 个网口  -> 单臂：该口作 lan，DHCP 客户端；60 秒拿不到地址则回落到
#               192.168.100.254/24（网关 192.168.100.1）
#   >=2 网口 -> 路由器：第一个口作 wan（DHCP/DHCPv6），其余桥接进 lan（192.168.100.1）
#
# rootfs 首启自动扩到整块虚拟磁盘（growroot），docker 数据在 /opt/docker。

#--- target ---
CONFIG_TARGET_armsr=y
CONFIG_TARGET_armsr_armv8=y
CONFIG_TARGET_DEVICE_armsr_armv8_DEVICE_generic=y

#--- image formats ---
# ext4 rootfs 默认仅 160 MiB（armsr 平台默认）。本镜像同时开启 EXT4FS 并装入
# docker / openclash / iStore / quickstart 等大量包，未压缩体积超出 160 MiB，
# 导致 make_ext4fs 报 "out of space"（RT28/R28S 只产 squashfs 不触发）。
# 放大到 512 MiB；虚拟机首启 growroot 会再扩到整块虚拟磁盘，仅影响初始镜像上限。
CONFIG_TARGET_ROOTFS_PARTSIZE=512
# ext4 = 默认（虚拟机里可扩容）；squashfs = 可一键恢复出厂
CONFIG_TARGET_ROOTFS_EXT4FS=y
CONFIG_TARGET_ROOTFS_SQUASHFS=y
CONFIG_TARGET_IMAGES_GZIP=y
CONFIG_TARGET_ROOTFS_INITRAMFS=n
