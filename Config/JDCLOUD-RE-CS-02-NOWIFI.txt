# 雅典娜 JDCloud AX6600 (RE-CS-02, IPQ6010) — NO WiFi
# Source : immortalwrt/immortalwrt @ master (official, device already present)
# Image  : immortalwrt-qualcommax-ipq60xx-jdcloud_re-cs-02-squashfs-sysupgrade.bin
#
# Scripts/Settings.sh disables kmod-ath11k/mac80211/wpad for *-NOWIFI*.
# The athena-led dot-matrix panel is still enabled (it is a GPIO display,
# completely independent of the wireless stack).

#--- target ---
CONFIG_TARGET_qualcommax=y
CONFIG_TARGET_qualcommax_ipq60xx=y
CONFIG_TARGET_DEVICE_qualcommax_ipq60xx_DEVICE_jdcloud_re-cs-02=y

#--- NAND tools ---
# 雅典娜是 SPI-NAND 机型。mtd-utils 的 nand-utils 带 `DEPENDS:=@NAND_SUPPORT`
# 平台门控（rockchip 无裸 NAND，放 GENERAL 会被静默丢弃），故只在本机型的配置里请求。
CONFIG_PACKAGE_nand-utils=y

#--- LED dot-matrix panel ---
# athena-led / luci-app-athena-led 来自第三方 (unraveloop/JDC-AX6600-Athena-LED-
# Controller v2.4.0, 见 Scripts/Packages.sh)。该 LuCI 包自带 i18n (.lmo 直接装进
# /usr/lib/lua/luci/i18n/), 不生成独立的 luci-i18n-athena-led-zh-cn 包。
CONFIG_PACKAGE_athena-led=y
CONFIG_PACKAGE_luci-app-athena-led=y
