# PonWrt (Airoha AN7581 / AN7583 ONT) — 全局默认值，拼在 PON 设备配置之前。
#   .config = Config/GENERAL-PON.txt + Config/PON-<SoC>.txt
# WRT-CORE 按配置名前缀选文件：配置名以 "PON-" 开头就用本文件，其它仍用
# Config/GENERAL.txt。
#
# 与 GENERAL.txt 的差异都是「设备上根本没有」而不是漏写：
#   · 没有 eMMC、没有独立数据分区 —— 所以没有 block-mount / automount /
#     mmc-utils / f2fs-tools / resize2fs / tune2fs / parted / sfdisk / gdisk
#     这一整条「找分区 → 格式化 → 挂到 /opt」的工具链，也没有
#     target-patch/{jdcloud,r28s,rt28}/opt-partition 那套首启脚本。
#     根分区是 UBI 里的 fit 卷，overlay 由 fstools 原生处理，不需要任何挂载。
#   · 不带 open-box（闭源面板，174 MB 解包到 /opt，此机型既没 /opt 也没空间）。
#   · 不带 docker：dockerd+containerd+runc+docker-compose 约 35 MB，加上基础包
#     镜像就 78–80 MB，NAND 上现成的 fit 卷往往没这么大（sysupgrade 不重排卷，
#     写完直接起不来）。要 docker 就在 Config/PON-*.txt 里把那段注释打开，
#     并且确认 fit 卷 ≥100 MB，或走 U-Boot「重建 UBI」。

#--- build options ---

CONFIG_TARGET_ROOTFS_INITRAMFS=y
CONFIG_TARGET_INITRAMFS_COMPRESSION_NONE=y

#--- kernel / netfilter ---
# PPE 硬件加速表：驱动侧由 target-patch/pon/931-*.patch 把 DRAM 表从 16K 扩到
# 64K 条（对齐原厂 hw_nat），表本体约 5.8MB 从 CMA 出，这里把 CMA 固定成
# 16MB，避免被其它驱动按需抢占后 dmam_alloc_coherent 失败。
CONFIG_CMA_SIZE_MBYTES=16

#--- IPTV: rtp2httpd（组播转单播）---

# 上面解决「软终端 / 手机 / 电视 App 直接播 HTTP」，任何 WiFi 都行。
#
# 「机顶盒要拿专网地址」这部分不再用 luci-app-iptv：
# 那个包只在 LuCI 里做一次性配置，刷机后还得手动点一遍，而且它的 zone /
# hotplug 又和 Inject-ITV.sh 里那套重叠。现在改成首启固化，见
# target-patch/pon/files/etc/uci-defaults/98-pon-net：
#   pon0.30 (VLAN 30) → br-iptv ← 让出的最后一个 LAN 口
#   接口 iptv 用 proto none，纯二层透传，盒子自己 DHCP / 拨号都行。
# 所以这里不再需要 kmod-nft-bridge / luci-app-iptv。

#--- WAN: DHCP / PPPoE 双协议 ---
# 实测 pon-26.11.23 的 .manifest：ppp 2.5.4 / ppp-mod-pppoe / luci-proto-ppp /
# odhcp6c / odhcpd-ipv6only 本来就在镜像里 —— airoha 虽然没显式写 DEVICE_TYPE，
# 但 include/target.mk 有 `DEVICE_TYPE?=router` 兜底，DEFAULT_PACKAGES.router 是
# 生效的，所以拨号能力一直都在。这里显式列出来，只是让 defconfig 结果可预期：
# 不依赖隐式默认值，将来上游若改兜底逻辑，这边也不会静默丢掉拨号能力。
# 用哪种由用户在「网络-接口-wan」里选：光猫桥接就 PPPoE，光猫已拨号就 DHCP。

CONFIG_PACKAGE_ppp-mod-pppoe=y
CONFIG_PACKAGE_luci-proto-ppp=y

CONFIG_PACKAGE_odhcpd-ipv6only=y
