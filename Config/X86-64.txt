# x86/64 — 通用 UEFI（EFI）虚拟机 / 物理机镜像
#
# 目标：ImmortalWrt 官方 x86/64（Generic EFI Boot），产物标准 GPT 磁盘：
#   p1 = FAT ESP（GRUB，BOOTX64.EFI）+ p2 = squashfs rootfs
# 在 KVM / VMware / Proxmox / 飞牛虚拟机 / 物理 x86 小主机上用 UEFI 固件直接启动，
# 不需要 U-Boot。固件首启会自动把根分区扩到整块虚拟磁盘（OpenWrt 自带 70-rootpt-resize）。
#
# 默认产出：immortalwrt-x86-64-generic-squashfs-combined-efi.img.gz（UEFI 虚拟机用）

#--- target ---
CONFIG_TARGET_x86=y
CONFIG_TARGET_x86_64=y
# x86/64 必须显式启用 generic 设备，否则 $(BUILD_DEVICES) 为空，image 循环不执行，
# 不会生成任何镜像与 profiles.json（曾导致 Compile Firmware 在 profiles.json 阶段失败）。
# 符号名由 scripts/target-metadata.pl 生成，格式是
#   TARGET_DEVICE_<target>_<subtarget>_DEVICE_<device-id>
# 且设备的默认值是 n（metadata.pm: $profile->{default} = "n"），所以必须写全、
# 写对才有效：x86/64 的 <target>_<subtarget> 是 x86_64，device-id 是 DEVICE_generic，
# 合起来就是 TARGET_DEVICE_x86_64_DEVICE_generic（对照 armsr 的
# CONFIG_TARGET_DEVICE_armsr_armv8_DEVICE_generic=y）。
CONFIG_TARGET_DEVICE_x86_64_DEVICE_generic=y
CONFIG_TARGET_ROOTFS_PARTSIZE=512
# squashfs = 可一键恢复出厂（虚拟机主用镜像）；不编 ext4 rootfs，少编一套、加快构建。
CONFIG_TARGET_ROOTFS_SQUASHFS=y
CONFIG_TARGET_IMAGES_GZIP=y
CONFIG_TARGET_ROOTFS_INITRAMFS=n
# 只保留虚拟机 / 物理机用的镜像与内核，去掉冗余 rootfs 归档。
CONFIG_TARGET_ROOTFS_TARGZ=n
CONFIG_TARGET_ROOTFS_CPIOGZ=n
# 显式开启 EFI 镜像（combined-efi），并直接保证 GRUB_EFI_IMAGES=y，
# 确保生成虚拟机用的 squashfs-combined-efi.img.gz。
CONFIG_GRUB_EFI_IMAGES=y

#--- GRUB / 控制台输出 ---
# x86 的 GRUB 会在 cmdline 里自动加 console=tty1（屏幕）和 console=ttyS0（串口），
# 末尾是串口 → /dev/console 指向 ttyS0，::{askconsole} 把 login 只开到串口。
# 屏幕因此只有内核日志、没有登录提示，看起来像“没画面/卡住”。
# 解决办法：在 Scripts/Settings.sh 的 x86 分支里给 inittab 追加 tty1 登录。
# 这里只需保证 GRUB 的 console 终端开启 + 菜单停留，方便看到启动过程。
CONFIG_GRUB_CONSOLE=y
# 菜单停留 3 秒，方便在控制台看到 GRUB 菜单（0 = 不等待直接启动）
CONFIG_GRUB_TIMEOUT="3"
